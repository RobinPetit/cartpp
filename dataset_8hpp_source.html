112 void save_to(const std::string& path) const {
113 std::ofstream outfile(
114 path,
115 std::ios::out | std::ios::binary | std::ios::app
116 );
117 // Header
118 char buffer[16] = {};
119 std::memcpy(buffer, "CART", 4);
120 buffer[4] = static_cast<char>(sizeof(Float));
121 buffer[5] = 1; // Version
122 outfile.write(buffer, 16);
123 // Data
124 _X.save_to(outfile);
125 _y.save_to(outfile);
126 _p.save_to(outfile);
127 _w.save_to(outfile);
128 size_t nb_covs{nb_cols};
129 outfile.write(AS_CONSTCHARPTR(&nb_covs), ssizeof(nb_covs));
130 for(const auto& modalities : _modalities) {
131 auto nb_modalities{modalities.size()};
132 outfile.write(AS_CONSTCHARPTR(&nb_modalities), ssizeof(nb_modalities));
133 for(const auto& modality : modalities) {
134 auto mod_size{static_cast<ssize_t>(modality.size())};
135 outfile.write(AS_CONSTCHARPTR(&mod_size), ssizeof(mod_size));
136 outfile.write(AS_CONSTCHARPTR(modality.data()), mod_size);

141 static Dataset<Float> load_from(const std::string& path) {
142 std::ifstream infile(path, std::ios::in | std::ios::binary);
143 char buffer[16] = {};
144 infile.read(buffer, 16);
145 if(std::strncmp(buffer, "CART", 4) != 0 or buffer[4] != sizeof(Float))
146 throw std::runtime_error("WRONG HEADER");
147 auto X{decltype(Dataset<Float>::_X)::load_from(infile)};
148 auto y{decltype(Dataset<Float>::_y)::load_from(infile)};
149 auto p{decltype(Dataset<Float>::_p)::load_from(infile)};
150 auto w{decltype(Dataset<Float>::_w)::load_from(infile)};
151 decltype(Dataset<Float>::__modalities) modalities;
152 size_t nb_covs;
153 infile.read(AS_CHARPTR(&nb_covs), ssizeof(nb_covs));
154 modalities.resize(nb_covs);
155 for(size_t i{0}; i < nb_covs; ++i) {
156 size_t nb_modalities;
157 infile.read(AS_CHARPTR(&nb_modalities), ssizeof(nb_modalities));
158 for(size_t j{0}; j < nb_modalities; ++j) {
159 ssize_t mod_size;
160 infile.read(AS_CHARPTR(&mod_size), ssizeof(mod_size));
161 modalities[i].emplace_back(mod_size+1, '\0');
162 infile.read(modalities[i].back().data(), mod_size);
163 }
164 }
165 return Dataset<Float>(
166 std::move(X), std::move(y), std::move(p),
167 std::move(w), std::move(modalities)
168 );
169 }

309 inline const auto& sorted_Xypw(size_t j) const {
310 auto& cache_sorted{const_cast<Dataset<Float>*>(this)->_cache_sorted};
311 auto& cached{const_cast<Dataset<Float>*>(this)->_cached};
312 if(cache_sorted.size() == 0) [[unlikely]] {
313 cached.resize(nb_features());
314 cache_sorted.resize(nb_features());
315 for(size_t j{0}; j < nb_features(); ++j)
316 cached[j] = false;
317 }
318 assert(_cache_sorted.size() > j);
319 if(not cached[j]) [[unlikely]] {
320 Array<Float> Xj{get_feature_vector(j, false)};
321 Sort::SortingAlgorithm method{
322 is_categorical(j)
323 ? Sort::SortingAlgorithm::QUICKSORT_3WAY
324 : Sort::SortingAlgorithm::MERGESORT
325 };
326 Array<size_t> sorted_indices{argsort(Xj, method)};
327 std::get<0>(cache_sorted[j]) = std::move(Xj[sorted_indices]);
328 std::get<1>(cache_sorted[j]) = std::move(_y[sorted_indices]);
329 std::get<2>(cache_sorted[j]) = std::move(_p[sorted_indices]);
330 if(is_weighted())
331 std::get<3>(cache_sorted[j]) = std::move(_w[sorted_indices]);
332 std::get<4>(cache_sorted[j]) = std::move(sorted_indices);
333 cached[j] = true;
334 }
335 return _cache_sorted[j];
336 }

455 inline std::pair<Dataset<Float>*, Dataset<Float>*> split_on(
456 int j, Float threshold) const {
457 assert(not is_categorical(j));
458 auto [Xj, y, p, w, indices] = sorted_Xypw(j);
459 auto it{std::lower_bound(Xj.cbegin(), Xj.cend(), threshold)};
460 assert(it != Xj.cend());
461 auto idx{static_cast<size_t>(std::distance(Xj.cbegin(), it))};
462 return {
463 at(indices.view(0, idx)),
464 at(indices.view(idx, indices.size()))
465 };

483 inline std::pair<Dataset<Float>*, Dataset<Float>*> split_on(
484 int j, uint64_t mask) const {
485 assert(is_categorical(j));
486 const Array<Float>& Xj{get_feature_vector(j)};
487 Array<bool> go_left(size(), false);
488 Array<bool> go_right(size(), false);
489 for(size_t i{0}; i < size(); ++i) {
490 uint64_t flag{1ull << static_cast<int>(Xj[i])};
491 if(mask & flag)
492 go_left[i] = true;
493 else
494 go_right[i] = true;
495 }
496 return {
497 at(go_left),
498 at(go_right)
