80 void fit(const Dataset<Float>& dataset) {
81 // TODO: handle fairness epsilon
82 auto start{std::chrono::system_clock::now()};
83 if(config.bootstrap) {
84 data = dataset.sample(
85 static_cast<size_t>(config.bootstrap_frac * dataset.size()),
86 config.bootstrap_replacement
87 );
88 owns_dataset = true;
89 } else {
90 data = &dataset;
91 owns_dataset = false;
92 }
93 is_categorical = Array<bool>(data->nb_features());
94 for(size_t j{0}; j < is_categorical.size(); ++j)
95 is_categorical[j] = data->is_categorical(j);
96 if(data->is_weighted())
97 prop_root_p0 = weighted_prop_false(data->get_p(), data->get_w());
98 else
99 prop_root_p0 = mean<bool, Float>(data->get_p());
100 // TODO: unclear what happens if config.exact_splits is set to false
101 if(config.exact_splits) {
102 for(size_t j{0}; j < data->nb_features(); ++j) {
103 if(not is_categorical[j])
104 continue;
105 auto nb_modalities{data->get_nb_unique_modalities(j)};
106 constexpr size_t max_nb_modalities{20};
107 if(nb_modalities > max_nb_modalities)
108 throw std::runtime_error(
109 std::to_string(nb_modalities) + " is too much for feature "
110 + std::to_string(j) + " (expected < "
111 + std::to_string(max_nb_modalities) + ")"
112 );
113 }
114 }
115 if(config.split_type == NodeSelector::DEPTH_FIRST) {
116 if constexpr(Loss::CanBeDepthFirst<LossType>::value) {
117 if(config.normalized_dloss) {
118 build_tree<NodeSelector::DEPTH_FIRST, true>(*data);
119 } else {
120 build_tree<NodeSelector::DEPTH_FIRST, false>(*data);
121 }
122 } else {
123 throw std::runtime_error("Cannot be depth first");
124 }
125 } else if(config.split_type == NodeSelector::BEST_FIRST) {
126 if constexpr(Loss::CanBeBestFirst<LossType>::value) {
127 if(config.normalized_dloss) {
128 build_tree<NodeSelector::BEST_FIRST, true>(*data);
129 } else {
130 build_tree<NodeSelector::BEST_FIRST, false>(*data);
131 }
132 } else {
133 throw std::runtime_error("Cannot be best first");
134 }
135 } else
136 throw std::runtime_error("Unknown NodeSelector");
137 if(root == nullptr) {
138 throw std::runtime_error("Fit failed. No root");
139 }
140 fitted = true;
141 auto end{std::chrono::system_clock::now()};
142 std::chrono::duration<double> elapsed{end-start};
143 if(config.verbose) {
144 std::cout << "Time elapsed: " << elapsed << "\n";
145 std::cout << nb_splitting_nodes << " nodes\n";
146 }
147 nb_features = data->nb_features();
148 if(owns_dataset) {
149 delete data;
150 data = nullptr;
151 }
152 for(const Node<Float>* node : *this) {
153 if(node->parent != nullptr and node->data != nullptr) {
154 delete node->data;
155 }
156 const_cast<Node<Float>*>(node)->data = nullptr;
157 }

179 inline Float _predict(const Array<Float>& X, size_t i, size_t n) const {
180 Node<Float>* node{root};
181 while(not node->is_leaf()) {
182 auto j{node->feature_idx};
183 if(is_categorical[j]) {
184 auto modality{1 << static_cast<int>(X[j*n + i])};
185 if(node->left_modalities & modality)
186 node = node->left_child;
187 else if(node->right_modalities & modality)
188 node = node->right_child;
189 else
190 break;
191 } else {
192 if(X[j*n + i] < node->threshold)
193 node = node->left_child;
194 else
195 node = node->right_child;
196 }
197 }
198 return node->pred;
199 }

250 template <NodeSelector selector, bool normalized_dloss=true>
251 void build_tree(const Dataset<Float>& dataset) {
252 Splitter<Float, LossType, selector, normalized_dloss> splitter(dataset, config);
253 do {
254 Node<Float>* node{splitter.split()};
255 if(node == nullptr)
256 break;
257 nodes.push_back(node);
258 if(config.verbose) {
259 std::cout << std::string(3*node->depth, ' ')
260 << "Node (" << node->id << "), Depth: " << node->depth
261 << ", Feature: " << node->feature_idx
262 << ", Threshold: " << node->threshold
263 << ", DLoss: " << std::scientific << node->dloss
264 << ", Loss: " << std::scientific << node->loss
265 << ", Mean_value: " << std::defaultfloat << node->pred
266 << ", N=" << node->nb_observations
267 << '\n';
268 }
269 if(root == nullptr)
270 root = node;
271 ++nb_splitting_nodes;
272 } while(nb_splitting_nodes < config.interaction_depth);
273 }

275 void _recalibrate(const Dataset<Float>* dataset, Node<Float>* node) {
276 if(dataset->size() == 0) {
277 assert(node != root);
278 node->pred = root->pred;
279 node->nb_observations = 0;
280 return;
281 }
282 if(dataset->is_weighted())
283 node->pred = weighted_mean(dataset->get_y(), dataset->get_w());
284 else
285 node->pred = mean<Float, Float>(dataset->get_y());
286 node->nb_observations = dataset->size();
287 if(not node->is_leaf()) {
288 Dataset<Float>* left_data{nullptr};
289 Dataset<Float>* right_data{nullptr};
290 if(dataset->is_categorical(node->feature_idx)) {
291 std::tie(left_data, right_data) = dataset->split_on(
292 node->feature_idx, node->left_modalities
293 );
294 } else {
295 std::tie(left_data, right_data) = dataset->split_on(
296 node->feature_idx, node->threshold
297 );
298 }
299 _recalibrate(left_data, node->left_child);
300 _recalibrate(right_data, node->right_child);
301 delete left_data;
302 delete right_data;
303 }
304 }
