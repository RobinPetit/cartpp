603template <std::floating_point Float>
604inline Float weighted_mean(
605 const Array<Float>& array, const Array<Float>& weights) {
606 if(array.size() == 0) [[unlikely]]
607 return Float(0);
608 Float sum{0};
609 Float den{0};
610 assert(array.size() == weights.size());
611 for(size_t i{0}; i < array.size(); ++i) {
612 sum += array[i]*weights[i];
613 den += weights[i];
614 }
615 return sum / den;
616}
